    function parseLocalDate(str){
      if(!str) return null;
      const [y,m,d] = str.split("-").map(Number);
      if(!y||!m||!d) return null;
      return new Date(y, m-1, d);
    }
    function todayLocal(){ const t=new Date(); return new Date(t.getFullYear(), t.getMonth(), t.getDate()); }
    function addDays(date, n){ const d=new Date(date); d.setDate(d.getDate()+n); return d; }
    function fmtDate(date){ return date.toLocaleDateString(undefined,{ year:"numeric", month:"short", day:"numeric" }); }
    function toISODate(date){ const y=date.getFullYear(); const m=String(date.getMonth()+1).padStart(2,"0"); const d=String(date.getDate()).padStart(2,"0"); return `${y}-${m}-${d}`; }

      if(!start) return null;

      if(diff < 1 || diff > TOTAL_DAYS) return null;
      return diff;
    }

    function shuffleWorkout(){
      const pool = filteredWorkouts();
      if(pool.length === 0){ toast("No workouts match those filters - try widening them."); return; }

      if(avail.length === 0){ avail = pool; reused = true; }
      const pick = avail[Math.floor(Math.random() * avail.length)];

    }

    function renderFavoritesRow(){
      if(libFavorites.length === 0){ $("libFavRow").innerHTML = `<span class="tiny">No favorites yet - tap the ☆ on any workout in the list below.</span>`; return; }
      $("libFavRow").innerHTML = libFavorites.map(id => {
        const w = WORKOUTS.find(x => x.id === id);
        if(!w) return "";
        const short = w.title.length > 28 ? w.title.slice(0,28) + "…" : w.title;
        return `<button class="history-chip" data-pick-fav="${id}">⭐ ${escapeHtml(short)}</button>`;
      }).join("");
      document.querySelectorAll("[data-pick-fav]").forEach(btn => btn.addEventListener("click", () => setCurrentPick(Number(btn.dataset.pickFav), false)));

    function renderHistoryRow(){
      const recent = prunedHistory()
        .filter(h => Date.now() - new Date(h.shownAt).getTime() <= 7*86400000)
        .sort((a,b) => new Date(b.shownAt) - new Date(a.shownAt));
      if(recent.length === 0){ $("libHistoryRow").innerHTML = `<span class="tiny">Nothing shuffled yet this week.</span>`; return; }
      const seen = new Set();
      const rows = [];
      recent.forEach(h => {
        if(seen.has(h.id)) return;
        seen.add(h.id);
        const w = WORKOUTS.find(x => x.id === h.id);
        if(!w) return;
        const short = w.title.length > 26 ? w.title.slice(0,26) + "…" : w.title;
        rows.push(`<span class="history-chip" title="${new Date(h.shownAt).toLocaleDateString()}" data-pick-fav="${w.id}">${escapeHtml(short)}</span>`);
      });
      $("libHistoryRow").innerHTML = rows.join("") || `<span class="tiny">Nothing shuffled yet this week.</span>`;
      document.querySelectorAll("#libHistoryRow [data-pick-fav]").forEach(chip => chip.addEventListener("click", () => setCurrentPick(Number(chip.dataset.pickFav), false)));
    }
    function renderNudge(){
      const recent = prunedHistory().sort((a,b) => new Date(b.shownAt) - new Date(a.shownAt)).slice(0,2);
      const bothHard = recent.length === 2 && recent.every(h => {
        const w = WORKOUTS.find(x => x.id === h.id);
        return w && (w.difficulty === "hard" || /high/i.test(w.impact));
      });
      if(!bothHard){ $("libNudge").innerHTML = ""; return; }
      $("libNudge").innerHTML = `<div class="nudge-banner"><span>💛 Two tough sessions in a row - an easy or recovery day could feel good.</span><button id="nudgeEasyBtn">Show me an easy pick</button></div>`;
      $("nudgeEasyBtn").addEventListener("click", () => {
        libFilters.difficulty = "easy"; libFilters.category = "all"; libFilters.equipment = "all";
        shuffleWorkout();
      });
    }
